const rooms = [
    {
        number: 101,
        type: "Standard",
        price: 2000,
        available: true
    },
    {
        number: 102,
        type: "Deluxe",
        price: 3000,
        available: false
    },
    {
        number: 103,
        type: "Suite",
        price: 5000,
        available: true
    }
];

Get available rooms:

const availableRooms = rooms.filter(
    room => room.available
);

Get their numbers:

const roomNumbers = availableRooms.map(
    room => room.number
);

Get their total price:

const total = availableRooms.reduce(
    (sum, room) => sum + room.price,
    0
);

So we're combining:

Array
 ↓
filter
 ↓
map
 ↓
reduce
